package com.momos.millenairejeim.jei.crafting.type;

import com.momos.millenairejeim.jei.crafting.type.base.IMillRecipe;
import com.momos.millenairejeim.util.MMLog;
import org.millenaire.culture.VillagerType;
import org.millenaire.goal.gathering.GatheringType;

import java.util.List;

/**
 * 动态配方解析管理器 {@link MillRecipeManager}。
 * 采用策略数组注册机制，无需写死 switch-case，方便后续轻松扩展新工艺。
 */
public class MillRecipeManager {

    /**
     * 策略解析器数组。
     * 后续若需要增加新的工艺类型，只需向该数组追加对应的 {@link IMillRecipe.IMillRecipeParser} 即可！
     */
    private static final IMillRecipe.IMillRecipeParser[] PARSERS = new IMillRecipe.IMillRecipeParser[] {
            new StandardCraftingRecipe.Parser(),
            new ResourceLootRecipe.Parser(),
            new CropHarvestRecipe.Parser(),
            new LivestockRecipe.Parser()
    };

    /**
     * 遍历解析器数组分发解析任务
     *
     * @param type 采集加工配置 {@link GatheringType}
     * @param villagers 绑定的村民列表
     * @return 解析生成的 {@link IMillRecipe}，若没有合适解析器或解析失败则返回 null
     */
    public static IMillRecipe parse(GatheringType type, List<VillagerType> villagers) {
        if (type == null || type.handlerId() == null) return null;

        String handlerId = type.handlerId().toLowerCase();

        // 遍历解析器数组，实现动态策略分发
        for (IMillRecipe.IMillRecipeParser parser : PARSERS) {
            if (parser.supports(handlerId, type)) {
                try {
                    return parser.parse(type, villagers);
                } catch (Exception e) {
                    MMLog.warn(
                            "[Millenaire-JEI] 配方解析异常: ID={}, Handler={}",
                            "[Millenaire-JEI] Exception parsing recipe: ID={}, Handler={}",
                            "[Millenaire-JEI] Exception lors de l'analyse de la recette : ID={}, Handler={}",
                            type.id(), handlerId
                    );
                    return null;
                }
            }
        }
        return null;
    }
}
